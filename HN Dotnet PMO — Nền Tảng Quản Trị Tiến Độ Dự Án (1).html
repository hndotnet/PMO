  id TEXT PRIMARY KEY,
  project TEXT NOT NULL,
  task TEXT NOT NULL,
  owner TEXT NOT NULL,
  start_date TEXT,
  due_date TEXT,
  progress INTEGER DEFAULT 0,
  status TEXT DEFAULT 'Đang làm',
  updated_at TEXT,
  note TEXT,
  images JSONB DEFAULT '[]'::jsonb
);
ALTER TABLE hndotnet_tasks ENABLE ROW LEVEL SECURITY;
